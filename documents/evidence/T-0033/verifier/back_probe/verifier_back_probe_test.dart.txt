import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';
import 'package:mc_fixtures_characters/mc_fixtures_characters.dart';
import 'package:mc_mocked_backend/mc_mocked_backend.dart';
import 'package:mini_client/main.dart' as app;

import 'e2e_support.dart';

// Verifier probe (not part of the repo): what Back does after the journey.
void main() {
  final binding = IntegrationTestWidgetsFlutterBinding.ensureInitialized();

  testWidgets('verifier back probe', (tester) async {
    final first = CharacterUniverse(42, defaultMockClock).byName.first;
    void report(String step) {
      debugPrint(
        'PROBE $step: '
        'home=${find.byKey(const Key('home.placeholder')).hitTestable().evaluate().length} '
        'grid=${find.byKey(charactersGridKey).hitTestable().evaluate().length} '
        'preview=${find.text('See full profile').hitTestable().evaluate().length} '
        'placeholder=${find.byKey(const Key('character.details.placeholder')).hitTestable().evaluate().length} '
        'backButtons=${find.byType(BackButton).hitTestable().evaluate().length}',
      );
    }

    startApp(app.main);
    await pumpUntil(tester, find.byKey(const Key('home.placeholder')), 'home');
    await tester.tap(find.text('Characters'));
    await pumpUntil(tester, find.text(first.name), 'list');
    report('list');
    await tester.tap(find.text(first.name));
    await pumpUntil(tester, find.text('See full profile'), 'preview');
    report('preview');
    await tester.tap(find.text('See full profile'));
    await pumpUntil(
      tester,
      find.byKey(const Key('character.details.placeholder')),
      'placeholder',
    );
    report('placeholder');

    await tester.tap(find.byType(BackButton).hitTestable().last);
    await letArtLoad(tester, const Duration(seconds: 2));
    report('after back from placeholder');
    await shot(binding, tester, 'probe_01_after_back_from_placeholder');

    final back = find.byType(BackButton).hitTestable();
    if (back.evaluate().isNotEmpty) {
      await tester.tap(back.last);
      await letArtLoad(tester, const Duration(seconds: 2));
      report('after second back');
      await shot(binding, tester, 'probe_02_after_second_back');
    }

    // System back (what Android's back key sends; iOS has none).
    final handled = await tester.binding.handlePopRoute();
    await letArtLoad(tester, const Duration(seconds: 2));
    debugPrint('PROBE system back handled=$handled');
    report('after system back');
    await shot(binding, tester, 'probe_03_after_system_back');
  });
}
