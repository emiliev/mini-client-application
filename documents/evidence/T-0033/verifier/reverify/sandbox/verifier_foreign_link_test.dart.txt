import 'package:flutter/services.dart';
import 'package:flutter/widgets.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';
import 'package:mc_e2e_utils/mc_e2e_utils.dart';
import 'package:mc_fixtures_characters/mc_fixtures_characters.dart';
import 'package:mc_mocked_backend/mc_mocked_backend.dart';
import 'package:universe_sandbox/main.dart' as app;

// Verifier probe (scratch copy only): a cold start with the MAIN app's scheme
// must be refused (the list opens), then warm links: a foreign one is refused,
// the sandbox's own one opens its route.
/// What the iOS embedder sends for a link while the app runs.
Future<void> _pushLink(WidgetTester tester, String location) =>
    tester.binding.defaultBinaryMessenger.handlePlatformMessage(
      'flutter/navigation',
      const JSONMethodCodec().encodeMethodCall(
        MethodCall('pushRouteInformation', {
          'location': location,
          'state': null,
        }),
      ),
      (_) {},
    );

void main() {
  final binding = IntegrationTestWidgetsFlutterBinding.ensureInitialized();

  testWidgets('verifier foreign links are refused', (tester) async {
    final first = CharacterUniverse(42, defaultMockClock).byName.first;
    const details = Key('character.details.placeholder');
    binding.platformDispatcher.defaultRouteNameTestValue =
        'miniclient://app/characters/5';
    startApp(app.main);
    await pumpUntil(tester, find.text(first.name), 'the list');
    expect(find.byKey(charactersGridKey), findsOneWidget);
    expect(find.byKey(details), findsNothing);
    debugPrint('PROBE cold foreign miniclient://: list shown, no details');
    await letArtLoad(tester, const Duration(seconds: 2));
    await shot(binding, tester, 'rv_02_sandbox_cold_foreign_link_list');

    for (final link in [
      'miniclient://app/characters/6',
      'https://example.com/characters/6',
      'miniclientuniverse://other/characters/6',
    ]) {
      await _pushLink(tester, link);
      await letArtLoad(tester, const Duration(seconds: 1));
      final shown = find.byKey(details).evaluate().length;
      debugPrint('PROBE warm foreign $link: details=$shown');
      expect(shown, 0, reason: link);
      expect(find.byKey(charactersGridKey), findsOneWidget, reason: link);
    }

    await _pushLink(tester, 'miniclientuniverse://app/characters/6');
    await pumpUntil(tester, find.byKey(details), 'details 6 (warm own link)');
    expect(find.text('Character details 6'), findsOneWidget);
    debugPrint('PROBE warm own-scheme: placeholder 6 shown');
    await shot(binding, tester, 'rv_03_sandbox_warm_own_link_details_6');
  });
}
