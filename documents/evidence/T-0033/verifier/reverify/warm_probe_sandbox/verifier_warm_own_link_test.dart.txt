// Verifier probe (scratch copy only): the sandbox's own link after boot, with
// no foreign link before it.
Future<void> _pushLink(WidgetTester tester, String location) =>
    tester.binding.defaultBinaryMessenger.handlePlatformMessage(
      'flutter/navigation',
      const JSONMethodCodec().encodeMethodCall(
        MethodCall('pushRouteInformation', {
          'location': location,
          'state': null,
        }),
      ),
      (_) {},
    );

void main() {
  final binding = IntegrationTestWidgetsFlutterBinding.ensureInitialized();

  testWidgets('verifier warm own-scheme link only', (tester) async {
    final first = CharacterUniverse(42, defaultMockClock).byName.first;
    const details = Key('character.details.placeholder');
    startApp(app.main);
    await pumpUntil(tester, find.text(first.name), 'the list');
    await letArtLoad(tester, const Duration(seconds: 2));
    String loc() => GoRouter.of(
      tester.element(find.byType(Navigator).first),
    ).state.uri.toString();
    debugPrint('PROBE before push: location=${loc()}');
    await _pushLink(tester, 'miniclientuniverse://app/characters/6');
    await letArtLoad(tester, const Duration(seconds: 3));
    debugPrint(
      'PROBE after own-scheme push: location=${loc()} '
      'details=${find.byKey(details).evaluate().length}',
    );
    await _pushLink(tester, '/characters/7');
    await letArtLoad(tester, const Duration(seconds: 3));
    debugPrint(
      'PROBE after plain-path push: location=${loc()} '
      'details=${find.byKey(details).evaluate().length} '
      'text7=${find.text('Character details 7').evaluate().length}',
    );
    await shot(binding, tester, 'rv_04_sandbox_warm_probe');
  });
}
