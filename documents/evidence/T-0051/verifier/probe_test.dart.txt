  seed: seed,

Dio dioOf(MockBackend b, {bool plain = false}) {
  final d = b.createDio();
  d.options.validateStatus = (_) => true;
  if (plain) d.options.responseType = ResponseType.plain;
  return d;
}

Future<String> signIn(Dio d, String email, String pw) async {
  final r = await d.post(
    '/v1/auth/sign-in',
    data: {'email': email, 'password': pw},
  );
  expect(r.statusCode, 200);
  return (r.data as Map)['accessToken'] as String;
}

Map<String, String> h(String t) => {'Authorization': 'Bearer $t'};
Future<Response> rq(Dio d, String m, String p, {Object? body, String? tok}) =>
    d.request(
      p,
      data: body,
      options: Options(method: m, headers: tok == null ? null : h(tok)),
    );

class P {
  P(this.id, this.cat, this.list, this.unit, this.stock);
  final int id, list, unit, stock;
  final String cat;
}

Future<List<P>> products(Dio d) async {
  final deals = <int, int>{};
  for (final x in (await d.get('/v1/store/deals')).data['deals']) {
    deals[x['product']['id']] = x['dealPriceCents'];
  }
  final out = <P>[];
  for (final c in (await d.get('/v1/store/categories')).data['categories']) {
    final r = await d.get('/v1/store/categories/${c['id']}/products?limit=100');
    for (final p in r.data['results']) {
      out.add(
        P(
          p['id'],
          p['categoryId'],
          p['priceCents'],
          deals[p['id']] ?? p['priceCents'],
          p['stock'],
        ),
      );
    }
  }
  return out;
}

// ---- my own oracle, written from CONTRACT.md only ----
const promoDefs = {
  'SAVE10': ('pct', 10, 0, ''),
  'SAVE25': ('pct', 25, 0, ''),
  'TAKE5': ('fix', 0, 500, ''),
  'TAKE15': ('fix', 0, 1500, ''),
  'HEROES3': ('b2g1', 0, 0, 'heroes'),
  'COSMIC3': ('b2g1', 0, 0, 'cosmic'),
};
const mins = {'TAKE5': 2500, 'TAKE15': 8000};
({
  List<int> lineTotals,
  int total,
  Map<String, int> perPromo,
  Map<String, bool> elig,
})
oracle(List<(P, int)> lines, List<String> codes) {
  final n = lines.length;
  final sub = [for (final l in lines) l.$1.unit * l.$2];
  final left = List<int>.from(sub);
  final per = <String, int>{};
  final elig = <String, bool>{};
  for (final c in codes.where((c) => promoDefs[c]!.$1 == 'b2g1')) {
    final cat = promoDefs[c]!.$4;
    final units = <List<int>>[];
    for (var i = 0; i < n; i++) {
      if (lines[i].$1.cat == cat) {
        for (var q = 0; q < lines[i].$2; q++) {
          units.add([lines[i].$1.unit, i]);
        }
      }
    }
    units.sort((a, b) => b[0] != a[0] ? b[0] - a[0] : a[1] - b[1]);
    var t = 0;
    for (var g = 0; g + 3 <= units.length; g += 3) {
      final u = units[g + 2];
      left[u[1]] -= u[0];
      t += u[0];
    }
    per[c] = t;
    elig[c] = units.length >= 3;
  }
  for (final c in codes.where((c) => promoDefs[c]!.$1 == 'pct')) {
    final pc = promoDefs[c]!.$2;
    var t = 0;
    for (var i = 0; i < n; i++) {
      final num = left[i] * pc;
      var cut = num ~/ 100;
      if (num % 100 >= 50) cut++;
      left[i] -= cut;
      t += cut;
    }
    per[c] = t;
    elig[c] = n > 0;
  }
  for (final c in codes.where((c) => promoDefs[c]!.$1 == 'fix')) {
    final r0 = left.fold(0, (a, b) => a + b);
    final a0 = promoDefs[c]!.$3;
    if (r0 >= mins[c]!) {
      final fl = [for (final r in left) a0 * r ~/ r0];
      final rest = a0 - fl.fold(0, (a, b) => a + b);
      final idx = List.generate(n, (i) => i)
        ..sort((a, b) {
          final ra = a0 * left[a] % r0, rb = a0 * left[b] % r0;
          return ra != rb ? rb - ra : a - b;
        });
      for (var k = 0; k < rest; k++) {
        fl[idx[k]]++;
      }
      for (var i = 0; i < n; i++) {
        left[i] -= fl[i];
      }
      per[c] = a0;
      elig[c] = true;
    } else {
      per[c] = 0;
      elig[c] = false;
    }
  }
  return (
    lineTotals: left,
    total: left.fold(0, (a, b) => a + b),
    perPromo: per,
    elig: elig,
  );
}

Future<void> check(
  String name,
  int seed,
  List<(P, int)> lines0,
  List<String> codes,
) async {
  final b = mk(seed);
  final d = dioOf(b);
  final byId = {for (final p in await products(d)) p.id: p};
  final lines = [for (final l in lines0) (byId[l.$1.id]!, l.$2)];
  final t = await signIn(d, 'reader@mini-client.test', 'marvelous');
  for (final l in lines) {
    final r = await rq(
      d,
      'PUT',
      '/v1/me/cart/items/${l.$1.id}',
      body: {'quantity': l.$2},
      tok: t,
    );
    expect(r.statusCode, 200, reason: '${r.data}');
  }
  for (final c in codes) {
    final r = await rq(
      d,
      'POST',
      '/v1/me/cart/promo',
      body: {'code': c},
      tok: t,
    );
    expect(r.statusCode, 200, reason: '$name $c ${r.data}');
  }
  final cart = (await rq(d, 'GET', '/v1/me/cart', tok: t)).data as Map;
  final o = oracle(lines, codes);
  final items = cart['items'] as List;
  expect([for (final i in items) i['totalCents']], o.lineTotals, reason: name);
  expect(cart['totalCents'], o.total, reason: name);
  expect(
    cart['totalCents'],
    items.fold<int>(0, (a, i) => a + (i['totalCents'] as int)),
  );
  for (final p in cart['promotions']) {
    expect(p['discountCents'], o.perPromo[p['code']], reason: '$name ${p['code']}');
    expect(p['eligible'], o.elig[p['code']], reason: '$name ${p['code']} elig');
  }
  // ignore: avoid_print
  print(
    '$name: total=${cart['totalCents']} lines=${o.lineTotals} promos=${[for (final p in cart['promotions']) '${p['code']}:${p['discountCents']}${p['eligible'] ? '' : '(inelig)'}']}',
  );
}
