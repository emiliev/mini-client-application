#!/bin/bash
# T-0056 Verifier mutations: 10 of the engineer's 69 re-run, plus 12 of my own.
# Usage: vmut.sh <worktree> <output_base> <log> [ids...]
# Each mutation edits one file, runs the targets, restores the file with
# `git checkout`, and checks the restore. KILLED needs "FAILED in"; a build
# failure is BROKEN (not a kill).
set -uo pipefail
WT="$1"; OB="$2"; LOG="$3"; shift 3
ONLY=" $* "
cd "$WT" || exit 2
say() { printf '%s\n' "$*" | tee -a "$LOG"; }
bz() { tool/bazelw --output_base="$OB" "$@"; }

FILES=("$C" "$S" "$PAGE" "$R" "$FF" "$FC" "$BLOC/featured_bloc.dart" "$BLOC/category_products_bloc.dart" "$WID/deals_section.dart")
restore_all() { git checkout -- "${FILES[@]}" 2>/dev/null; }
trap restore_all EXIT INT TERM
TARGETS="//feature/store/store_home:bdd_test //feature/store/store_home:flow_scope_test //feature/store/store_home:contract_test //app/mini_client:analytics_screens_test //app/mini_client:store_route_test //utility/mc_feature_flow:feature_flow_test //utility/mc_analytics:flow_observer_test //utility/mc_analytics:route_observer_test"
run() { bz test --nocache_test_results --test_timeout=900 $TARGETS 2>&1 | tr '\r' '\n'; }

if [[ " $ONLY " == *" BASE "* ]]; then
  out="$(run)"
  if grep -q -a -E "FAILED|NO STATUS|Build did NOT complete" <<< "$out"; then
    say "BASELINE FAILS"; grep -a -E "FAILED|Build did NOT" <<< "$out" | tee -a "$LOG"; exit 1
  fi
  say "baseline: PASS ($(grep -a -c 'PASSED in' <<< "$out") targets)"
fi

mutate() {
  local id="$1" what="$2" file="$3"; shift 3
  if [[ "$ONLY" != *" $id "* ]]; then return; fi
  /usr/bin/python3 -I - "$file" "$@" <<'EOF'
import sys
path, pairs = sys.argv[1], sys.argv[2:]
s = open(path).read()
for old, new in zip(pairs[0::2], pairs[1::2]):
    if s.count(old) != 1:
        print("PATTERN MATCHES %d TIMES: %r" % (s.count(old), old)); sys.exit(3)
    s = s.replace(old, new)
open(path, "w").write(s)
EOF
  if [[ $? -ne 0 ]]; then say "$id $what: NOT APPLIED"; git checkout -- "$file"; return; fi
  fvm dart format "$file" > /dev/null
  local out; out="$(run)"
  if grep -q -a -E "Build did NOT complete|FAILED TO BUILD" <<< "$out"; then
    say "$id $what: BROKEN"; grep -a -E "error|Error" <<< "$out" | head -5 | tee -a "$LOG"
  elif grep -q -a "FAILED in" <<< "$out"; then
    say "$id $what: KILLED ($(grep -a 'FAILED in' <<< "$out" | awk '{print $1}' | paste -sd' ' -))"
  elif grep -q -a "TIMEOUT in" <<< "$out"; then
    say "$id $what: TIMEOUT"
  else
    say "$id $what: SURVIVED"
  fi
  git checkout -- "$file"
  git diff --quiet -- "$file" || say "RESTORE FAILED: $file"
}
