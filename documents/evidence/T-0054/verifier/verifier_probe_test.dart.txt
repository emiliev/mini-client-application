class _Flow extends StatelessWidget {
  const _Flow({required this.onComplete, required this.label});
  final ValueChanged<String?> onComplete;
  final String label;
  @override
  Widget build(BuildContext context) => TextButton(
    key: _finish,
    onPressed: () => onComplete(label),
    child: Text('finish $label'),
  );
}

class _H {
  _H(this.router, this.signedIn, this.tick);
  final GoRouter router;
  final ValueNotifier<bool> signedIn;
  final ValueNotifier<int> tick;
  final List<String> got = [];
  void create() => unawaited(
    AppFlows.createReadingList(router).then((v) => got.add('c:$v')),
  );
  void edit(String id) => unawaited(
    AppFlows.editReadingList(router, id).then((v) => got.add('e:$v')),
  );
}

int _changes = 0;

Future<_H> _boot(
  WidgetTester tester, {
  bool signedIn = true,
  bool asyncGuard = false,
  int guardMs = 50,
}) async {
  final state = ValueNotifier<bool>(signedIn);
  final tick = ValueNotifier<int>(0);
  addTearDown(state.dispose);
  addTearDown(tick.dispose);
  FutureOr<String?> guard(BuildContext c, GoRouterState s) {
    String? target() {
      if (!state.value && s.uri.path.startsWith('/me/')) {
        return '/signin?from=${Uri.encodeComponent(s.uri.toString())}';
      }
      return null;
    }

    if (asyncGuard && s.uri.path.startsWith('/me/')) {
      return Future<String?>.delayed(
        Duration(milliseconds: guardMs),
        target,
      );
    }
    return target();
  }

  final router = GoRouter(
    extraCodec: const FlowExtraCodec(),
    refreshListenable: Listenable.merge([state, tick]),
    redirect: guard,
    routes: [
      GoRoute(path: '/', builder: (c, s) => const Text('home')),
      GoRoute(path: '/other', builder: (c, s) => const Text('other')),
      GoRoute(
        path: '/signin',
        builder: (c, s) => Column(
          children: [
            TextButton(
              key: _signIn,
              onPressed: () {
                state.value = true;
                GoRouter.of(c).go(s.uri.queryParameters['from']!);
              },
              child: const Text('sign in'),
            ),
            TextButton(
              key: _signInPop,
              onPressed: () => Navigator.of(c).pop('popped'),
              child: const Text('pop'),
            ),
          ],
        ),
      ),
      GoRoute(
        path: '/me/lists/new',
        builder: (c, s) => FlowResultRoute<String>(
          extra: s.extra,
          builder: (complete) => _Flow(onComplete: complete, label: 'new-id'),
        ),
      ),
      GoRoute(
        path: '/me/lists/:id/edit',
        builder: (c, s) => FlowResultRoute<String>(
          extra: s.extra,
          builder: (complete) =>
              _Flow(onComplete: complete, label: s.pathParameters['id']!),
        ),
      ),
    ],
  );
  addTearDown(router.dispose);
  await tester.pumpWidget(MaterialApp.router(routerConfig: router));
  await tester.pumpAndSettle();
  return _H(router, state, tick);
}

/// Router changes after every call ended reach no watch.
Future<void> _noListener(WidgetTester tester, _H h) async {
  final before = _changes;
  h.router.go('/other');
  await tester.pumpAndSettle();
  h.router.go('/');
  await tester.pumpAndSettle();
  expect(_changes, before, reason: 'a listener is left on the delegate');
}

Future<void> _systemBack(WidgetTester tester) async {
  await tester.binding.handlePopRoute();
  await tester.pumpAndSettle();
}

Future<void> _restore(WidgetTester tester, GoRouter router) async {
  final saved = router.routeInformationParser.restoreRouteInformation(
    router.routerDelegate.currentConfiguration,
  )!;
  // Through JSON, as the engine would.
  final state = jsonDecode(jsonEncode(saved.state));
  await tester.binding.defaultBinaryMessenger.handlePlatformMessage(
    'flutter/navigation',
    const JSONMethodCodec().encodeMethodCall(
      MethodCall('pushRouteInformation', {
        'location': saved.uri.toString(),
        'state': state,
      }),
    ),
    (_) {},
  );
  await tester.pumpAndSettle();
}

Future<void> _tapFinish(WidgetTester tester) async {
  await tester.tap(find.byKey(_finish));
  await tester.pumpAndSettle();
}
