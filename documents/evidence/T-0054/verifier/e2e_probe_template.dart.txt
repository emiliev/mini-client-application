// T-0054 Verifier device probe (lives only in the e2e copy, never the repo).
// COLD_LINK, BACK_OR_HOME and SHOT are substituted per target.
import 'dart:io';

import 'package:app_core/app_core.dart';
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:go_router/go_router.dart';
import 'package:integration_test/integration_test.dart';
import 'package:mc_e2e_utils/mc_e2e_utils.dart';
import 'package:mini_client/main.dart' as app;

const _home = Key('home.placeholder');
const _notFound = Key('not.found');
const _goHome = Key('not.found.home');

int _depth(WidgetTester tester) => GoRouter.of(
  tester.element(find.byKey(_notFound)),
).routerDelegate.currentConfiguration.matches.length;

Future<void> _leave(WidgetTester tester, bool back) async {
  await tester.tap(back ? find.byType(BackButton) : find.byKey(_goHome));
  await pumpUntil(tester, find.byKey(_home), 'home after one action');
  expect(find.byKey(_notFound), findsNothing);
  expect(find.byType(BackButton), findsNothing, reason: 'home is bottom');
}

void main() {
  final binding = IntegrationTestWidgetsFlutterBinding.ensureInitialized();

  testWidgets('VE cold COLD_LINK then a running /nope', (tester) async {
    File('${Directory.systemTemp.path}/$e2eStartLinkFile').writeAsStringSync(
      'miniclient://appCOLD_LINK',
    );
    startApp(app.main);
    await pumpUntil(tester, find.byKey(_notFound), 'the not-found page');
    expect(find.text('Nothing lives at COLD_LINK'), findsOneWidget);
    expect(find.byType(BackButton), findsOneWidget);
    expect(find.byKey(_goHome), findsOneWidget);
    expect(_depth(tester), 2, reason: '[home, not-found]');
    await shot(binding, tester, 'SHOT_a_cold');
    await _leave(tester, BACK_OR_HOME);
    await shot(binding, tester, 'SHOT_b_home');

    // A running link to /nope from the characters list: [home, not-found].
    await tester.tap(find.text('Characters'));
    await tester.pumpAndSettle();
    await pushLink(tester, 'miniclient://app/nope');
    await pumpUntil(tester, find.byKey(_notFound), 'running /nope');
    expect(find.text('Nothing lives at /nope'), findsOneWidget);
    expect(_depth(tester), 2, reason: '[home, not-found] while running');
    await shot(binding, tester, 'SHOT_c_running_nope');
    await _leave(tester, !BACK_OR_HOME);
    await shot(binding, tester, 'SHOT_d_home');
  });
}
