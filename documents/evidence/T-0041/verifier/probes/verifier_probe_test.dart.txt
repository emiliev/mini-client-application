// T-0041 Verifier probe (scratch copy only, never committed to the code).
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';
import 'package:mc_e2e_utils/mc_e2e_utils.dart';
import 'package:mc_fixtures_characters/mc_fixtures_characters.dart';
import 'package:mc_fixtures_comics/mc_fixtures_comics.dart';
import 'package:mc_mocked_backend/mc_mocked_backend.dart';
import 'package:mini_client/main.dart' as app;

const _comicTitle = ValueKey<String>('comics_details.title');
const _signInSheet = Key('publishing.sign_in.sheet');
const _signInConfirm = Key('publishing.sign_in.confirm');

Future<void> _back(WidgetTester tester) async {
  final back = find.byType(BackButton).hitTestable();
  expect(back, findsWidgets, reason: 'a back button is on screen');
  await tester.tap(back.last);
  await tester.pump(const Duration(milliseconds: 800));
}

void main() {
  final binding = IntegrationTestWidgetsFlutterBinding.ensureInitialized();

  testWidgets('verifier probe: back navigation, sheet drag while signing in, '
      'two pages of one comic', (tester) async {
    final characters = CharacterUniverse(42, defaultMockClock).byName;
    final first = characters.first;
    final comic = ComicUniverse(42).comicsOfCharacter(first.id).first;

    startApp(app.main);
    await pumpUntil(tester, find.byKey(const Key('home.placeholder')), 'home');
    await tester.tap(find.text('Characters'));
    await pumpUntil(tester, find.text(first.name), 'the first card');
    await tester.tap(find.text(first.name));
    await pumpUntil(tester, find.text('Comics'), 'related comics');
    await tester.scrollUntilVisible(
      find.text(comic.title),
      200,
      scrollable: find.byType(Scrollable).last,
    );
    await tester.pump(const Duration(milliseconds: 600));
    await tester.tap(find.text(comic.title));
    await pumpUntil(tester, find.byKey(_comicTitle), 'the comic page');

    // 1. Back: comic page -> preview -> list.
    await _back(tester);
    await pumpUntil(
      tester,
      find.byKey(characterPreviewTitleKey).hitTestable(),
      'the preview after Back',
    );
    expect(find.byKey(_comicTitle).hitTestable(), findsNothing);
    await shot(binding, tester, 'v01_back_to_preview');
    await _back(tester);
    await pumpUntil(
      tester,
      find.byKey(charactersGridKey).hitTestable(),
      'the list after Back',
    );
    expect(find.byKey(characterPreviewTitleKey).hitTestable(), findsNothing);
    await shot(binding, tester, 'v02_back_to_list');

    // 2. The sheet: drag it down while signing in.
    await tester.tap(find.text(first.name));
    await pumpUntil(tester, find.text('Comics'), 'related comics again');
    await tester.scrollUntilVisible(
      find.text(comic.title),
      200,
      scrollable: find.byType(Scrollable).last,
    );
    await tester.pump(const Duration(milliseconds: 600));
    await tester.tap(find.text(comic.title));
    await pumpUntil(tester, find.byKey(_comicTitle), 'the comic page again');
    await tester.tap(find.text('Add to favourites'));
    await pumpUntil(tester, find.byKey(_signInSheet), 'the sign-in sheet');
    await tester.tap(find.byKey(_signInConfirm));
    await tester.pump(const Duration(milliseconds: 16));
    // Back while busy: refused.
    await tester.binding.handlePopRoute();
    await tester.pump(const Duration(milliseconds: 100));
    final afterBack = find.byKey(_signInSheet).evaluate().isNotEmpty;
    // ignore: avoid_print
    print('VERIFIER sheet present after Back while busy: $afterBack');
    await tester.drag(find.byKey(_signInSheet), const Offset(0, 600));
    await tester.pump(const Duration(milliseconds: 600));
    final afterDrag = find.byKey(_signInSheet).evaluate().isNotEmpty;
    // ignore: avoid_print
    print('VERIFIER sheet present after drag while busy: $afterDrag');
    await shot(binding, tester, 'v03_after_drag_while_signing_in');
    await pumpUntil(
      tester,
      find.text('Remove from favourites'),
      'the favourite saved even though the sheet was dragged',
    );
    expect(find.byKey(_signInSheet), findsNothing);
    await shot(binding, tester, 'v04_favourite_after_drag');

    // 3. Two pages of the same comic: a link to it over itself.
    await pushLink(tester, 'miniclient://app/comics/${comic.id}');
    await tester.pump(const Duration(seconds: 2));
    await pumpUntil(
      tester,
      find.text('Remove from favourites').hitTestable(),
      'the second page of the same comic',
    );
    await tester.tap(find.text('Remove from favourites').hitTestable());
    await pumpUntil(
      tester,
      find.text('Add to favourites').hitTestable(),
      'removed on the top page',
    );
    await _back(tester);
    await tester.pump(const Duration(seconds: 1));
    final lowerSaysRemove = find
        .text('Remove from favourites')
        .hitTestable()
        .evaluate()
        .isNotEmpty;
    // ignore: avoid_print
    print('VERIFIER lower page still says Remove (stale): $lowerSaysRemove');
    await shot(binding, tester, 'v05_lower_page_after_removal_on_top');
  });
}
