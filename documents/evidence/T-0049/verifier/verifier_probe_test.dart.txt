// T-0049 Verifier probes (scratch, not committed).
import 'dart:io';

import 'package:app_core/app_core.dart';
import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:go_router/go_router.dart';
import 'package:mc_di/mc_di.dart';
import 'package:mc_monitoring/mc_monitoring.dart';
import 'package:mc_session/mc_session.dart';
import 'package:mc_test_utils/mc_test_utils.dart';
import 'package:mini_client/src/mini_client_app.dart';
import 'package:mini_client/src/pages/placeholder_pages.dart';

const _config =
    '{"env":"e2e","backend":{"seed":42,"latencyMs":0,"scenario":"happy"},'
    '"apiBaseUrl":"https://mock.mini-client.invalid/","build":"1"}';

const _home = Key('home.placeholder');
const _details = ValueKey<String>('reading_list_editor.details');
const _grid = ValueKey<String>('characters_list.grid');
const _comicTitle = ValueKey<String>('comics_details.title');
const _notFound = Key('not.found');

ValueKey<String> _result(int id) =>
    ValueKey<String>('reading_list_editor.result.$id');

Future<void> _pushLink(WidgetTester tester, String location) =>
    tester.binding.defaultBinaryMessenger.handlePlatformMessage(
      'flutter/navigation',
      const JSONMethodCodec().encodeMethodCall(
        MethodCall('pushRouteInformation', {
          'location': location,
          'state': null,
        }),
      ),
      (_) {},
    );

void main() {
  late Directory storage;

  setUp(() async {
    await resetDIContainer();
    storage = Directory.systemTemp.createTempSync('verifier_probe_test');
  });

  tearDown(() async {
    await resetDIContainer();
    storage.deleteSync(recursive: true);
  });

  Future<void> boot(WidgetTester tester, {String? coldStartLink}) async {
    tester.view
      ..devicePixelRatio = 1
      ..physicalSize = mcReferenceSurface;
    addTearDown(tester.view.reset);
    final links = PendingLinks.install(tester.binding);
    addTearDown(links.detach);
    if (coldStartLink != null) {
      links.offer(coldStartLink);
    }
    await tester.pumpWidget(
      DefaultAssetBundle(
        bundle: McFixtureAssetBundle(),
        child: MiniClientApp(
          links: links,
          initializers: (routerRef) => rootInitializers(
            config: () async => _config,
            storageDirectory: storage,
            monitoringSinks: [InMemorySink()],
            mockDomains: appMockDomains,
            routerRef: routerRef,
            wirings: wiringInitializers,
          ),
        ),
      ),
    );
    await tester.pump();
    await tester.pumpAndSettle();
  }

  Future<void> signInOnTheSheet(WidgetTester tester) async {
    expect(find.byKey(fixtureSignInSheetKey), findsOneWidget);
    await tester.tap(find.byKey(fixtureSignInButtonKey));
    for (var i = 0; i < 20; i++) {
      await tester.runAsync(
        () => Future<void>.delayed(const Duration(milliseconds: 20)),
      );
      await tester.pumpAndSettle();
      if (find.byKey(fixtureSignInSheetKey).evaluate().isEmpty) {
        break;
      }
    }
    expect(diContainer<SessionReader>().isSignedIn, isTrue);
  }

  Future<void> tap(WidgetTester tester, Finder finder) async {
    await tester.ensureVisible(finder);
    await tester.pumpAndSettle();
    await tester.tap(finder.hitTestable());
    await tester.pump(const Duration(milliseconds: 500));
    await tester.pumpAndSettle();
  }

  GoRouter router(WidgetTester tester) =>
      GoRouter.of(tester.element(find.byType(Scaffold).first));

  /// The page stack as the outer navigator sees it (page names / types).
  List<String> stack(WidgetTester tester) {
    final nav = tester.state<NavigatorState>(find.byType(Navigator).first);
    return nav.widget.pages.map((p) => '${p.name}').toList();
  }

  Future<void> backOnce(WidgetTester tester) async {
    expect(find.byType(BackButton), findsWidgets, reason: 'a back button');
    await tester.tap(find.byType(BackButton).first);
    await tester.pumpAndSettle();
  }

  Future<void> expectHomeAtBottom(WidgetTester tester) async {
    expect(find.byKey(_home), findsOneWidget);
    expect(find.byType(BackButton), findsNothing);
    expect(router(tester).canPop(), isFalse);
  }

  // ---------- AC1 extra cold links ----------
  testWidgets('VPROBE cold /me/lists/l-1/edit: back once to home', (
    tester,
  ) async {
    await boot(tester, coldStartLink: 'miniclient://app/me/lists/l-1/edit');
    await signInOnTheSheet(tester);
    expect(find.text('Edit reading list'), findsOneWidget);
    // ignore: avoid_print
    print('VPROBE stack edit: ${stack(tester)}');
    await backOnce(tester);
    await expectHomeAtBottom(tester);
  });

  testWidgets('VPROBE cold /search?q=spi#top: search with spi, back home', (
    tester,
  ) async {
    await boot(tester, coldStartLink: 'miniclient://app/search?q=spi#top');
    final field = tester.widget<EditableText>(
      find.descendant(
        of: find.byKey(const ValueKey<String>('discovery_search.field')),
        matching: find.byType(EditableText),
      ),
    );
    // ignore: avoid_print
    print(
      'VPROBE query+fragment field="${field.controller.text}" '
      'uri=${router(tester).state.uri}',
    );
    expect(field.controller.text, 'spi');
    await backOnce(tester);
    await expectHomeAtBottom(tester);
  });

  testWidgets('VPROBE cold /comics/abc: what is under not-found', (
    tester,
  ) async {
    await boot(tester, coldStartLink: 'miniclient://app/comics/abc');
    expect(find.byKey(_notFound), findsOneWidget);
    // ignore: avoid_print
    print(
      'VPROBE comics/abc stack=${stack(tester)} '
      'back=${find.byType(BackButton).evaluate().length} '
      'canPop=${router(tester).canPop()}',
    );
  });

  testWidgets('VPROBE cold /nope (unknown path)', (tester) async {
    await boot(tester, coldStartLink: 'miniclient://app/nope');
    expect(find.byKey(_notFound), findsOneWidget);
    // ignore: avoid_print
    print(
      'VPROBE nope stack=${stack(tester)} '
      'back=${find.byType(BackButton).evaluate().length} '
      'canPop=${router(tester).canPop()}',
    );
  });

  testWidgets('VPROBE cold /creators/1 nests too', (tester) async {
    await boot(tester, coldStartLink: 'miniclient://app/creators/1');
    // ignore: avoid_print
    print('VPROBE creators stack=${stack(tester)}');
    await backOnce(tester);
    await expectHomeAtBottom(tester);
  });

  // ---------- AC2 from inside flows / sheets ----------
  testWidgets('VPROBE link from the characters preview (inner flow page)', (
    tester,
  ) async {
    await boot(tester);
    await tap(tester, find.text('Characters'));
    // Open the first card's preview.
    final card = find.descendant(
      of: find.byKey(_grid),
      matching: find.byType(InkWell),
    );
    await tap(tester, card.first);
    expect(find.text('See full profile'), findsOneWidget, reason: 'preview');
    // ignore: avoid_print
    print('VPROBE preview before link stack=${stack(tester)}');

    await _pushLink(tester, 'miniclient://app/comics/501500');
    await tester.pumpAndSettle();
    expect(find.byKey(_comicTitle), findsOneWidget);
    // ignore: avoid_print
    print('VPROBE preview after link stack=${stack(tester)}');
    expect(stack(tester), hasLength(2));
    await backOnce(tester);
    await expectHomeAtBottom(tester);
  });

  testWidgets('VPROBE link from reading-list step 2; then New works', (
    tester,
  ) async {
    await boot(tester);
    await tap(tester, find.byKey(newReadingListKey));
    await signInOnTheSheet(tester);
    await tester.enterText(find.byType(TextField).first, 'Amber reads');
    await tester.pumpAndSettle();
    await tap(tester, find.text('Next'));
    expect(find.byKey(_result(500060)), findsOneWidget, reason: 'step 2');

    await _pushLink(tester, 'miniclient://app/search?q=spi');
    await tester.pumpAndSettle();
    // ignore: avoid_print
    print('VPROBE step2 after link stack=${stack(tester)}');
    expect(stack(tester), hasLength(2));
    await backOnce(tester);
    await expectHomeAtBottom(tester);

    await tap(tester, find.byKey(newReadingListKey));
    expect(find.byKey(_details), findsOneWidget, reason: 'New works again');
    expect(find.text('Step 1 of 3'), findsOneWidget);
  });

  testWidgets('VPROBE link while the sign-in sheet is open; then New works', (
    tester,
  ) async {
    await boot(tester);
    await tap(tester, find.byKey(newReadingListKey));
    expect(find.byKey(fixtureSignInSheetKey), findsOneWidget);

    await _pushLink(tester, 'miniclient://app/comics/501500');
    await tester.pumpAndSettle();
    expect(find.byKey(fixtureSignInSheetKey), findsNothing);
    expect(find.byKey(_comicTitle), findsOneWidget);
    // ignore: avoid_print
    print('VPROBE sheet after link stack=${stack(tester)}');
    expect(stack(tester), hasLength(2));
    await backOnce(tester);
    await expectHomeAtBottom(tester);

    await tap(tester, find.byKey(newReadingListKey));
    expect(
      find.byKey(fixtureSignInSheetKey),
      findsOneWidget,
      reason: 'New opens the editor (asking to sign in) again',
    );
  });

  // ---------- AC3 D2 repro (T-0048 verifier probe) and Edit ----------
  testWidgets('VPROBE D2: after a link replaced the open editor, New opens '
      'an editor again', (tester) async {
    await boot(tester);
    await tap(tester, find.byKey(newReadingListKey));
    await signInOnTheSheet(tester);
    expect(find.text('Step 1 of 3'), findsOneWidget);
    await _pushLink(tester, 'miniclient://app/');
    await tester.pumpAndSettle();
    expect(find.byKey(_home), findsOneWidget);
    expect(find.byKey(_details), findsNothing);
    await tap(tester, find.byKey(newReadingListKey));
    expect(find.text('Step 1 of 3'), findsOneWidget);
  });

  testWidgets('VPROBE Edit works again after a link replaced the editor; '
      'and a result still reaches home', (tester) async {
    await boot(tester);
    await tap(tester, find.byKey(newReadingListKey));
    await signInOnTheSheet(tester);
    await tester.enterText(find.byType(TextField).first, 'Amber reads');
    await tester.pumpAndSettle();
    await tap(tester, find.text('Next'));
    await tap(tester, find.byKey(_result(500060)));
    await tap(tester, find.text('Next'));
    await tap(tester, find.text('Create list'));
    expect(find.text('List created: l-5'), findsOneWidget);

    await tap(tester, find.byKey(editReadingListKey));
    expect(find.text('Edit reading list'), findsOneWidget);
    await _pushLink(tester, 'miniclient://app/characters');
    await tester.pumpAndSettle();
    expect(find.byKey(_grid), findsOneWidget);
    await backOnce(tester);
    await expectHomeAtBottom(tester);
    expect(
      find.text('List created: l-5'),
      findsOneWidget,
      reason: 'null result leaves the last list',
    );

    await tap(tester, find.byKey(editReadingListKey));
    expect(find.text('Edit reading list'), findsOneWidget, reason: 'Edit');
    await tap(tester, find.text('Next'));
    await tap(tester, find.text('Next'));
    await tap(tester, find.text('Save changes'));
    expect(find.text('List saved: l-5'), findsOneWidget);
  });

  // ---------- AC2 control: push semantics would differ ----------
  testWidgets('VPROBE an in-app push still stacks on what is open', (
    tester,
  ) async {
    await boot(tester);
    await tap(tester, find.text('Characters'));
    final _ = router(tester).push<void>(AppLocations.comic(501500));
    await tester.pumpAndSettle();
    // ignore: avoid_print
    print('VPROBE push stack=${stack(tester)}');
    expect(stack(tester), hasLength(3));
  });
}
