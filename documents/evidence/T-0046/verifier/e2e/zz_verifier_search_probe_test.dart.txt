import 'dart:io';

import 'package:app_core/app_core.dart';

import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';
import 'package:mc_e2e_utils/mc_e2e_utils.dart';
import 'package:mc_fixtures_discovery/mc_fixtures_discovery.dart';
import 'package:mc_mocked_backend/mc_mocked_backend.dart';
import 'package:mini_client/main.dart' as app;

// T-0046 Verifier probe (temporary, scratch copy only).
const _field = ValueKey<String>('discovery_search.field');
const _results = ValueKey<String>('discovery_search.results');
const _noHistory = ValueKey<String>('discovery_search.history.none');
const _history = ValueKey<String>('discovery_search.history');
const _comicTitle = ValueKey<String>('comics_details.title');

EditableText _editable(WidgetTester tester) => tester.widget<EditableText>(
  find.descendant(of: find.byKey(_field), matching: find.byType(EditableText)),
);

void main() {
  final binding = IntegrationTestWidgetsFlutterBinding.ensureInitialized();

  testWidgets('V-AC5 blank-query start link, then a link, a comic, the real '
      'Back button and the results', (tester) async {
    final index = SearchIndex.fromUniverses(42, defaultMockClock);
    final comics = index.search('spi', types: ['comic']).toList();
    final comic = comics[1].entry;
    File('${Directory.systemTemp.path}/$e2eStartLinkFile')
        .writeAsStringSync('miniclient://app/search?q=');

    startApp(app.main);
    await pumpUntil(tester, find.byKey(_field), 'the search page from ?q=');
    await tester.pump(const Duration(seconds: 1));
    expect(find.byKey(const Key('home.placeholder')), findsNothing);
    expect(_editable(tester).controller.text, '');
    expect(_editable(tester).focusNode.hasFocus, isTrue, reason: 'autofocus');
    expect(find.byKey(_results), findsNothing);
    expect(
      find.byKey(_noHistory).evaluate().isNotEmpty ||
          find.byKey(_history).evaluate().isNotEmpty,
      isTrue,
      reason: 'the recent searches (or the hint) show',
    );
    await shot(binding, tester, 'v01_blank_query_link');

    await pushLink(tester, 'miniclient://app/search?q=spi');
    await pumpUntil(
      tester,
      find
          .descendant(
            of: find.byKey(_results),
            matching: find.text('Spindle Alchemist'),
          )
          .hitTestable(),
      'the results of the second link',
    );
    FocusManager.instance.primaryFocus?.unfocus();
    await tester.pump(const Duration(seconds: 1));
    await tester.tap(
      find.descendant(
        of: find.byKey(const ValueKey<String>('discovery_search.filters')),
        matching: find.text('Comics'),
      ),
    );
    await tester.pump(const Duration(seconds: 1));
    final row = find.descendant(
      of: find.byKey(_results),
      matching: find.text(comic.title),
    );
    await pumpUntil(tester, row.hitTestable(), 'the second comic row');
    await tester.tap(row.hitTestable());
    await pumpUntil(tester, find.byKey(_comicTitle), 'the comic page');
    expect(tester.widget<Text>(find.byKey(_comicTitle)).data, comic.title);

    // The real Back button of the comic page, not router.pop().
    final back = find.byType(BackButton).hitTestable();
    expect(back, findsOneWidget, reason: 'the comic page has a Back button');
    await tester.tap(back);
    await pumpUntil(tester, row.hitTestable(), 'the results again after Back');
    expect(find.byKey(_comicTitle), findsNothing);
    expect(_editable(tester).controller.text, 'spi');
    await letArtLoad(tester, const Duration(seconds: 2));
    await shot(binding, tester, 'v02_back_to_results');
  });
}
